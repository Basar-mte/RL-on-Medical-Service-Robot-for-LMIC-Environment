cmake_minimum_required(VERSION 3.16)
project(hospitalnav_gz_plugins)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-plugin REQUIRED COMPONENTS register)
find_package(gz_transport_vendor REQUIRED)
find_package(gz-transport REQUIRED)
find_package(gz_msgs_vendor REQUIRED)
find_package(gz-msgs REQUIRED)
set(CMAKE_CXX_STANDARD 17)
add_library(hospitalnav_pedestrian_bodies SHARED src/PedestrianBodies.cc)
target_link_libraries(hospitalnav_pedestrian_bodies gz-sim::gz-sim gz-plugin::register
  gz-transport::gz-transport gz-msgs::gz-msgs)
add_library(hospitalnav_replay_rig SHARED src/ReplayRig.cc)
target_link_libraries(hospitalnav_replay_rig gz-sim::gz-sim gz-plugin::register)
install(TARGETS hospitalnav_pedestrian_bodies hospitalnav_replay_rig LIBRARY DESTINATION lib)
# make gz find the plugin whenever this workspace is sourced
ament_environment_hooks("${CMAKE_CURRENT_SOURCE_DIR}/hooks/gz_plugin_path.dsv.in")
ament_package()
